**Step 3: Update `src/rpc/router.cpp`**
`Response::id` and `ErrorResponse::id` still use `nlohmann::json`. In dispatch:
- When building responses, assign `req->id.value_or(nullptr)`
- Notifications (no id): return empty string (don't send response)

```cpp
std::string Router::dispatch(const std::string& raw_message, const Context& ctx) {
  auto req = Request::parse(raw_message);
  if (!req.has_value()) {
    ErrorResponse err;
    err.id = nullptr;
    err.code = error::PARSE_ERROR;
    err.message = "Parse error";
    return err.serialize();
  }

  bool is_notification = !req->id.has_value();

  auto it = handlers_.find(req->method);
  if (it == handlers_.end()) {
    if (is_notification) return "";
    ErrorResponse err;
    err.id = req->id.value_or(nullptr);
    err.code = error::METHOD_NOT_FOUND;
    err.message = "Method not found: " + req->method;
    return err.serialize();
  }

  try {
    auto result = it->second(ctx, req->params);
    if (is_notification) return "";
    Response resp;
    resp.id = req->id.value_or(nullptr);
    resp.result = std::move(result);
    return resp.serialize();
  } catch (const RpcError& e) {
    if (is_notification) return "";
    ErrorResponse err;
    err.id = req->id.value_or(nullptr);
    err.code = e.code;
    err.message = e.message;
    return err.serialize();
  } catch (const nlohmann::json::exception& e) {
    if (is_notification) return "";
    ErrorResponse err;
    err.id = req->id.value_or(nullptr);
    err.code = error::INVALID_PARAMS;
    err.message = std::string("Invalid params: ") + e.what();
    return err.serialize();
  } catch (const std::exception& e) {
    if (is_notification) return "";
    ErrorResponse err;
    err.id = req->id.value_or(nullptr);
    err.code = error::INTERNAL_ERROR;
    err.message = e.what();
    return err.serialize();
  }
}
```

```cpp
  if (wk.environment == entt::null || !reg.valid(wk.environment))
    throw rpc::RpcError{error::ENTITY_NOT_FOUND, "No environment entity"};
  auto& env = reg.get<Environment>(wk.environment);
  return nlohmann::json{{"env", serializeEnvironment(env)}};
});
```

## Appendix: CMake E2E deps problem

The E2E test binary links the full game engine but the existing `CMakeLists.txt` compiles everything into one executable. The cleanest solution without a full refactor:

1. Keep `file(GLOB_RECURSE SOURCE "src/*.cpp")` for the main executable
2. For E2E tests, glob `src/game/*.cpp src/rpc/*.cpp src/utils/*.cpp` (exclude `src/app/*.cpp`, `src/editors/*.cpp`, `src/main.cpp`)
3. Link against the same `DEPS_SOURCES` variable from `cmake/deps.cmake`
4. This avoids a full library refactor while still excluding SFML-heavy editor code

The `src/game/draw_engine.cpp` and `src/game/draw_manager.cpp` use SFML — either exclude them or add `#ifdef HEADLESS_MODE` guards around SFML-specific code in those files.
